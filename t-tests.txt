# Re-import needed packages due to kernel reset
import pandas as pd
import numpy as np
from scipy.stats import ttest_ind

pd.set_option('display.max_rows', 200)
def collect_welch_tests_by_strategy(df_all, metrics, baseline_strategy="5RR"):
    results = []

    strategies = df_all["Strategy"].unique()
    strategies = [s for s in strategies if s != baseline_strategy]

    for strategy in strategies:
        df_base = df_all[df_all["Strategy"] == baseline_strategy]
        df_compare = df_all[df_all["Strategy"] == strategy]

        for metric in metrics:
            try:
                x = df_compare[metric].dropna()
                y = df_base[metric].dropna()

                # Welch's t-test
                t_stat, p_val = ttest_ind(x, y, equal_var=False)

                # Cohen's d
                nx, ny = len(x), len(y)
                dof = nx + ny - 2
                pooled_std = np.sqrt(((nx - 1) * np.var(x, ddof=1) + (ny - 1) * np.var(y, ddof=1)) / dof)
                d = (np.mean(x) - np.mean(y)) / pooled_std

                # Descriptive stats
                mean_x = np.mean(x)
                mean_y = np.mean(y)
                is_latency = "Latency" in metric
                is_better = (mean_x < mean_y) if is_latency else (mean_x > mean_y)
                percent_diff = ((mean_x - mean_y) / mean_y) * 100

                # Interpretations
                if p_val < 0.001:
                    p_comment = "Very significant (*** p < 0.001)"
                elif p_val < 0.01:
                    p_comment = "Significant (** p < 0.01)"
                elif p_val < 0.05:
                    p_comment = "Significant (* p < 0.05)"
                elif p_val < 0.1:
                    p_comment = "Weak evidence (p < 0.1)"
                else:
                    p_comment = "Not significant (p ≥ 0.1)"

                if abs(d) < 0.2:
                    magnitude = "negligible effect"
                elif abs(d) < 0.5:
                    magnitude = "small effect"
                elif abs(d) < 0.8:
                    magnitude = "medium effect"
                else:
                    magnitude = "large effect"
                direction = "PLB improves" if (d < 0 if is_latency else d > 0) else "PLB degrades"
                effect_comment = f"{direction} ({magnitude})"

                results.append({
                    "Compared Strategy": strategy,
                    "Metric": metric,
                    "t-stat": round(t_stat, 4),
                    f"{strategy} mean": round(mean_x, 4),
                    f"{baseline_strategy} mean": round(mean_y, 4),
                    "p-value": "{:.4e}".format(p_val),
                    "p-value comment": p_comment,
                    "PLB < RR": mean_x < mean_y,
                    "PLB better?": is_better,
                    "Cohen's d": round(d, 4),
                    "Effect comment": effect_comment,
                    "Percent change": f"{percent_diff:+.2f}%"
                })

            except Exception as e:
                results.append({
                    "Compared Strategy": strategy,
                    "Metric": metric,
                    "t-stat": "ERROR",
                    f"{strategy} mean": None,
                    f"{baseline_strategy} mean": None,
                    "p-value": str(e),
                    "p-value comment": "N/A",
                    "PLB < RR": None,
                    "PLB better?": None,
                    "Cohen's d": None,
                    "Effect comment": "N/A",
                    "Percent change": "N/A"
                })

    return pd.DataFrame(results)

def collect_welch_tests_by_strategy_and_terminals(df_all, metrics, baseline_strategy="5RR"):
    results = []

    strategies = df_all["Strategy"].unique()
    strategies = [s for s in strategies if s != baseline_strategy]
    terminal_counts = sorted(df_all["Terminals"].unique())

    for terminals in terminal_counts:
        for strategy in strategies:
            df_base = df_all[(df_all["Strategy"] == baseline_strategy) & (df_all["Terminals"] == terminals)]
            df_compare = df_all[(df_all["Strategy"] == strategy) & (df_all["Terminals"] == terminals)]

            for metric in metrics:
                try:
                    x = df_compare[metric].dropna()
                    y = df_base[metric].dropna()

                    if len(x) < 2 or len(y) < 2:
                        raise ValueError("Not enough data points for t-test")

                    # Welch's t-test
                    t_stat, p_val = ttest_ind(x, y, equal_var=False)

                    # Cohen's d
                    nx, ny = len(x), len(y)
                    dof = nx + ny - 2
                    pooled_std = np.sqrt(((nx - 1) * np.var(x, ddof=1) + (ny - 1) * np.var(y, ddof=1)) / dof)
                    d = (np.mean(x) - np.mean(y)) / pooled_std

                    mean_x = np.mean(x)
                    mean_y = np.mean(y)
                    is_latency = "Latency" in metric
                    is_better = (mean_x < mean_y) if is_latency else (mean_x > mean_y)
                    percent_diff = ((mean_x - mean_y) / mean_y) * 100

                    # Interpretations
                    if p_val < 0.001:
                        p_comment = "Very significant (*** p < 0.001)"
                    elif p_val < 0.01:
                        p_comment = "Significant (** p < 0.01)"
                    elif p_val < 0.05:
                        p_comment = "Significant (* p < 0.05)"
                    elif p_val < 0.1:
                        p_comment = "Weak evidence (p < 0.1)"
                    else:
                        p_comment = "Not significant (p ≥ 0.1)"

                    if abs(d) < 0.2:
                        magnitude = "negligible effect"
                    elif abs(d) < 0.5:
                        magnitude = "small effect"
                    elif abs(d) < 0.8:
                        magnitude = "medium effect"
                    else:
                        magnitude = "large effect"

                    direction = "Improves" if (d < 0 if is_latency else d > 0) else "Degrades"
                    effect_comment = f"{direction} ({magnitude})"

                    results.append({
                        "Terminals": terminals,
                        "Compared Strategy": strategy,
                        "Metric": metric,
                        "t-stat": round(t_stat, 4),
                        f"{strategy} mean": round(mean_x, 4),
                        f"{baseline_strategy} mean": round(mean_y, 4),
                        "p-value": "{:.4e}".format(p_val),
                        "p-value comment": p_comment,
                        "Strategy < RR": mean_x < mean_y,
                        "Strategy better?": is_better,
                        "Cohen's d": round(d, 4),
                        "Effect comment": effect_comment,
                        "Percent change": f"{percent_diff:+.2f}%"
                    })

                except Exception as e:
                    results.append({
                        "Terminals": terminals,
                        "Compared Strategy": strategy,
                        "Metric": metric,
                        "t-stat": "ERROR",
                        f"{strategy} mean": None,
                        f"{baseline_strategy} mean": None,
                        "p-value": str(e),
                        "p-value comment": "N/A",
                        "Strategy < RR": None,
                        "Strategy better?": None,
                        "Cohen's d": None,
                        "Effect comment": "N/A",
                        "Percent change": "N/A"
                    })

    return pd.DataFrame(results)
